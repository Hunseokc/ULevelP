```tsx
// frontend/src/components/common/GeminiKeyOnboardingModal.tsx
import React, { useEffect, useState } from 'react';
import { apiClient, IntegrationDto } from '../../lib/apiClient';
import { SafeLink } from './SafeLink';
import styles from './GeminiKeyOnboardingModal.module.css';

export const GEMINI_STUDIO_URL = 'https://aistudio.google.com/app/apikey';
export const GEMINI_KEY_REGEX = /^AIza[0-9A-Za-z\-_]{35}$/;

export interface GeminiKeyValidationResult {
  valid: boolean;
  trimmed: string;
  status: 'idle' | 'ok' | 'err';
  hint: string;
}

/** Gemini API 키 문자열을 trim()하고 39자리 형식(^AIza[0-9A-Za-z-_]{35}$)을 실시간 검사한다 */
export function validateGeminiKeyFormat(rawInput: string | null | undefined): GeminiKeyValidationResult {
  const trimmed = String(rawInput ?? '').trim();
  if (!trimmed) {
    return {
      valid: false,
      trimmed: '',
      status: 'idle',
      hint: 'AIza로 시작하는 39자리 API 키를 붙여넣어 주세요.',
    };
  }
  if (GEMINI_KEY_REGEX.test(trimmed)) {
    return {
      valid: true,
      trimmed,
      status: 'ok',
      hint: '✓ 형식이 올바릅니다 (39자)',
    };
  }
  return {
    valid: false,
    trimmed,
    status: 'err',
    hint: `AIza로 시작하는 39자리 키 전체를 복사해 주세요 (현재 ${trimmed.length}자)`,
  };
}

export interface GeminiKeyOnboardingModalProps {
  open: boolean;
  initialModel?: string;
  onClose: () => void;
  onSaved?: (dto: IntegrationDto) => void;
}

export function GeminiKeyOnboardingModal({
  open,
  initialModel = 'gemini-flash-latest',
  onClose,
  onSaved,
}: GeminiKeyOnboardingModalProps): React.ReactElement | null {
  const [rawKey, setRawKey] = useState('');
  const [model, setModel] = useState(initialModel || 'gemini-flash-latest');
  const [showKey, setShowKey] = useState(false);
  const [saving, setSaving] = useState(false);
  const [resultMsg, setResultMsg] = useState('');
  const [resultKind, setResultKind] = useState<'' | 'ok' | 'err'>('');

  useEffect(() => {
    if (open) {
      setRawKey('');
      setModel(initialModel || 'gemini-flash-latest');
      setShowKey(false);
      setSaving(false);
      setResultMsg('');
      setResultKind('');
    }
  }, [open, initialModel]);

  useEffect(() => {
    if (!open) return;
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && !saving) {
        onClose();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [open, saving, onClose]);

  if (!open) return null;

  const validation = validateGeminiKeyFormat(rawKey);

  const handleSave = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!validation.valid) {
      setResultKind('err');
      setResultMsg(validation.hint);
      return;
    }
    if (!apiClient.isLoggedIn()) {
      setResultKind('err');
      setResultMsg('로그인 후 Gemini API 키를 저장할 수 있어요.');
      return;
    }

    setSaving(true);
    setResultKind('');
    setResultMsg('Gemini API 키 유효성을 실시간으로 확인하는 중…');

    try {
      const dto = await apiClient.integrations.saveGemini({
        apiKey: validation.trimmed,
        model: model.trim() || 'gemini-flash-latest',
        verify: true,
      });
      const modelLabel = dto.displayName ? ` · ${dto.displayName}` : '';
      setResultKind('ok');
      setResultMsg(
        `✓ 연결 완료! 무료 Gemini 모델이 정상 작동합니다 (끝 4자리 ${dto.hint || ''}${modelLabel})`
      );
      setRawKey('');
      if (onSaved) {
        onSaved(dto);
      }
    } catch (err: any) {
      setResultKind('err');
      setResultMsg((err && err.message) || 'Gemini API 키 검증 및 저장에 실패했어요.');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div
      className={styles.backdrop}
      id="geminiOnboardingModal"
      role="dialog"
      aria-modal="true"
      aria-labelledby="gemModalTitle"
      onClick={(e) => {
        if (e.target === e.currentTarget && !saving) onClose();
      }}
    >
      <div className={styles.modal}>
        <div className={styles.head}>
          <div>
            <h2 className={styles.title} id="gemModalTitle">
              AI 기능을 무료로 이용하기 위한 1분 설정
            </h2>
            <p className={styles.sub}>
              Google의 무료 Gemini API 키를 연결하면 모든 영상 분석 기능을 제한 없이 이용하실 수 있습니다.
            </p>
          </div>
          <button
            type="button"
            className={styles.closeBtn}
            id="btnCloseGeminiModal"
            onClick={onClose}
            disabled={saving}
          >
            닫기 ✕
          </button>
        </div>

        <div className={styles.badges} aria-label="안심 포인트">
          <span className={styles.reassureBadge}>✓ 무료 사용 가능 (분당 15회 무료 요청)</span>
          <span className={styles.reassureBadge}>✓ 신용카드 등록 불필요</span>
          <span className={styles.reassureBadge}>
            🔒 AES-256-GCM 암호화 보관 (브라우저 localStorage에 키 원문 미저장)
          </span>
        </div>

        <div className={styles.stepsBox}>
          <ol className={styles.stepsList}>
            <li>
              <b>1단계 (Google AI Studio 접속):</b> 아래 버튼을 눌러 기존 구글 계정으로 로그인합니다.
            </li>
            <li>
              <b>2단계 (API 키 생성):</b> 화면 좌측 상단 또는 중앙의 <b>[Create API key]</b> 파란색 버튼을 누릅니다.
            </li>
            <li>
              <b>3단계 (프로젝트 선택):</b> 팝업창에서 <b>[Create API key in new project]</b>(새 프로젝트에서 키 만들기)를 선택합니다.
            </li>
            <li>
              <b>4단계 (키 복사 및 붙여넣기):</b> 생성된 <code>AIza...</code>로 시작하는 긴 문자열 옆의 <b>[Copy]</b>를 누른 뒤 아래 입력창에 붙여넣어 주세요.
            </li>
          </ol>
          <div className={styles.studioRow}>
            <SafeLink
              href={GEMINI_STUDIO_URL}
              external
              className="ghostBtn"
              id="btnOpenAiStudio"
            >
              Google AI Studio 키 발급 페이지 바로 열기 ↗
            </SafeLink>
          </div>
        </div>

        <form className={styles.fieldGroup} onSubmit={handleSave}>
          <label className={styles.fieldLabel} htmlFor="gemModalKeyInput">
            복사한 Gemini API 키 붙여넣기 (AIza...)
          </label>
          <div className={styles.inputRow}>
            <input
              id="gemModalKeyInput"
              className="inpField"
              type={showKey ? 'text' : 'password'}
              placeholder="AIza..."
              autoComplete="off"
              value={rawKey}
              onChange={(e) => setRawKey(e.target.value.trim())}
              disabled={saving}
            />
            <button
              type="button"
              className={styles.toggleBtn}
```

```tsx

import React, { useEffect, useRef, useState } from 'react';
import { AppShell } from '../components/layout/AppShell';
import { SafeLink } from '../components/common/SafeLink';
import { GeminiKeyOnboardingModal } from '../components/common/GeminiKeyOnboardingModal';
import { auth, useAuth } from '../lib/auth';
import { AnalysisJobDto, apiClient, NoteSummaryDto } from '../lib/apiClient';
import { raw, safeUrl, thumb, videoIdFrom } from '../lib/security';
import styles from './NotesPage.module.css';

export const NOTE_FORM_TAGS: ReadonlyArray<string> = [
  '화성',
  '사운드 디자인',
  'DAW/MIDI',
  '리듬',
  '편곡',
  '믹싱',
  'TAB',
];

export const NOTE_FILTER_TAGS: ReadonlyArray<string> = ['', ...NOTE_FORM_TAGS];

function formatAgo(iso?: string | null): string {
  if (!iso) return '';
  const diffSec = (Date.now() - new Date(iso).getTime()) / 1000;
  if (diffSec < 60) return '방금';
  if (diffSec < 3600) return `${Math.floor(diffSec / 60)}분 전`;
  if (diffSec < 86400) return `${Math.floor(diffSec / 3600)}시간 전`;
  return new Date(iso).toLocaleDateString('ko-KR');
}

export function NotesPage(): React.ReactElement {
  const authState = useAuth();
  const activeUid = authState.user?.id ?? null;

  const [urlVal, setUrlVal] = useState('');
  const [selectedTags, setSelectedTags] = useState<string[]>([]);
  const [memoVal, setMemoVal] = useState('');
  const [formMsg, setFormMsg] = useState<{ text: string; kind: '' | 'ok' | 'err' }>({ text: '', kind: '' });
  const [submitting, setSubmitting] = useState(false);

  const [jobs, setJobs] = useState<AnalysisJobDto[]>([]);
  const [notes, setNotes] = useState<NoteSummaryDto[]>([]);
  const [filterTag, setFilterTag] = useState('');
  const [onboardingOpen, setOnboardingOpen] = useState(false);
  const [localHasKeyOverride, setLocalHasKeyOverride] = useState(false);

  const reqSeqRef = useRef(0);
  const pollTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const activeJobIdsRef = useRef<Set<string>>(new Set());

  const checkHasGeminiKey = (): boolean => {
    if (!apiClient.isLoggedIn()) return false;
    if (localHasKeyOverride) return true;
    const cached = apiClient.getCachedKeys();
    if (cached && cached.gemini) return true;
    const me = auth.getMe();
    if (me && me.keys && me.keys.gemini) return true;
    return false;
  };

  const hasGemini = checkHasGeminiKey();
  const loggedIn = Boolean(activeUid && apiClient.isLoggedIn());
  const pickedVideoId = videoIdFrom(urlVal);
  const notionLinkUrl = safeUrl(raw.get('ulp:notion-notes'));

  // 초기 URL 파라미터(?url=...) 복원
  useEffect(() => {
    const q = new URLSearchParams(window.location.search);
    const initialUrl = q.get('url');
    if (initialUrl) {
      setUrlVal(initialUrl);
      window.history.replaceState(null, '', window.location.pathname);
    }
  }, []);

  const loadNotesOnly = async (tag: string, uid: string | null) => {
    if (!uid || !apiClient.isLoggedIn()) {
      setNotes([]);
      return;
    }
    const mySeq = ++reqSeqRef.current;
    try {
      const list = await apiClient.notes.list(tag);
      if (mySeq !== reqSeqRef.current || auth.getUser()?.id !== uid) return;
      setNotes(Array.isArray(list) ? list : []);
    } catch {
      /* ignore background error */
    }
  };

  const refreshJobsAndNotes = async (uid: string | null, tag: string) => {
    if (pollTimerRef.current) {
      clearTimeout(pollTimerRef.current);
      pollTimerRef.current = null;
    }
    if (!uid || !apiClient.isLoggedIn()) {
      setJobs([]);
      setNotes([]);
      return;
    }

    const mySeq = ++reqSeqRef.current;
    try {
      const fetchedJobs = await apiClient.notes.listJobs();
      if (mySeq !== reqSeqRef.current || auth.getUser()?.id !== uid) return;

      const safeJobs = Array.isArray(fetchedJobs) ? fetchedJobs : [];
      setJobs(safeJobs);

      const hasInvalidKeyJob = safeJobs.some(
        (j) => j && j.status === 'failed' && typeof j.error === 'string' && j.error.startsWith('PROVIDER_KEY_INVALID:')
      );
      if (hasInvalidKeyJob) {
        setLocalHasKeyOverride(false);
      }

      const hasActive = safeJobs.some((j) => j && (j.status === 'queued' || j.status === 'running'));
      const newlyFinished = safeJobs.some((j) => j && j.status === 'done' && activeJobIdsRef.current.has(j.id));
      activeJobIdsRef.current = new Set(
        safeJobs.filter((j) => j && (j.status === 'queued' || j.status === 'running')).map((j) => j.id)
      );

      if (newlyFinished || !safeJobs.some((j) => activeJobIdsRef.current.has(j.id))) {
        const fetchedNotes = await apiClient.notes.list(tag);
        if (mySeq !== reqSeqRef.current || auth.getUser()?.id !== uid) return;
        setNotes(Array.isArray(fetchedNotes) ? fetchedNotes : []);
      }

      if (hasActive) {
        pollTimerRef.current = setTimeout(() => {
          void refreshJobsAndNotes(uid, tag);
        }, 5000);
      }
    } catch {
      /* ignore background error */
    }
  };

  useEffect(() => {
    setLocalHasKeyOverride(false);
    activeJobIdsRef.current.clear();
    void refreshJobsAndNotes(activeUid, filterTag);
    return () => {
      if (pollTimerRef.current) {
        clearTimeout(pollTimerRef.current);
      }
    };
  }, [activeUid]);

  const toggleFormTag = (tag: string) => {
    setSelectedTags((prev) => (prev.includes(tag) ? prev.filter((t) => t !== tag) : [...prev, tag]));
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!pickedVideoId) {
      setFormMsg({ text: '유튜브 영상 링크가 아니에요.', kind: 'err' });
      return;
    }
    if (selectedTags.length === 0) {
      setFormMsg({ text: '태그를 하나 이상 골라주세요.', kind: 'err' });
      return;
    }
    if (!checkHasGeminiKey()) {
      setFormMsg({ text: 'Gemini API 키가 필요해요. 1분 무료 키 발급 모달 또는 SETTING에서 등록해 주세요.', kind: 'err' });
      return;
    }

    setSubmitting(true);
    setFormMsg({ text: '분석 요청을 보내는 중...', kind: '' });
    try {
      const orderedTags = NOTE_FORM_TAGS.filter((t) => selectedTags.includes(t));
      await apiClient.notes.analyze({
        url: `https://www.youtube.com/watch?v=${pickedVideoId}`,
        tags: orderedTags,
        memo: memoVal.trim(),
      });
      setFormMsg({ text: '분석을 요청했어요. 보통 1~3분 뒤에 정리가 완료돼요.', kind: 'ok' });
      setUrlVal('');
      setMemoVal('');
      void refreshJobsAndNotes(activeUid, filterTag);
    } catch (err: any) {
      if (err && (err.code === 'NO_KEY' || err.status === 412)) {
        setLocalHasKeyOverride(false);
      }
      setFormMsg({ text: (err && err.message) || '요청 실패', kind: 'err' });
    } finally {
      setSubmitting(false);
    }
  };
```

```tsx
export function parseTimestampSec(ts: unknown): number | null {
  if (!ts || typeof ts !== 'string') return null;
  const parts = ts.trim().match(/\d+/g);
  if (!parts || parts.length < 2 || parts.length > 3) return null;
  let total = 0;
  for (const p of parts) {
    const num = parseInt(p, 10);
    if (Number.isNaN(num)) return null;
    total = total * 60 + num;
  }
  return total;
}
```
