   ```tsx
   import React, { useState } from 'react';
   import styles from './AppShell.module.css';
   import { useAuth, auth } from '../../lib/auth';
   import { getTheme, setTheme } from '../../lib/security';

   export type AppShellVariant = 'default' | 'narrow' | 'player';
   export type ActiveNavKey = 'home' | 'chords' | 'practice' | 'log' | 'notes' | 'settings' | 'privacy';

   export interface AppShellProps {
     variant?: AppShellVariant;
     activeNav?: ActiveNavKey;
     showThemeToggle?: boolean;
     children: React.ReactNode;
   }

   export function AppShell({
     variant = 'default',
     activeNav = 'home',
     showThemeToggle = false,
     children,
   }: AppShellProps): React.ReactElement {
     const authState = useAuth();
     const [theme, setThemeState] = useState<'light' | 'dark' | ''>(() => getTheme());

     const signedInUser = authState.user;
     const navLabel = signedInUser ? auth.computeInitial(signedInUser, authState.me) : 'LOGIN';
     const navAriaLabel = signedInUser
       ? `계정 설정 (${signedInUser.email || navLabel})`
       : 'Google 계정으로 로그인';
     const embedded = auth.isEmbedded();

     const handleAuthClick = async (e: React.MouseEvent<HTMLAnchorElement>) => {
       if (signedInUser) return;
       if (embedded) return;
       if (!auth.isConfigured()) return;
       e.preventDefault();
       try {
         await auth.signIn();
       } catch {
         window.location.href = 'settings.html#account';
       }
     };

     const handleThemePick = (next: 'light' | 'dark') => {
       const nextVal = theme === next ? '' : next;
       setTheme(nextVal);
       setThemeState(nextVal);
     };

     const shellClass = variant === 'player' ? styles.playerShell : styles.shell;
     const mainClass =
       variant === 'narrow'
         ? styles.mainNarrow
         : variant === 'player'
           ? styles.mainPlayer
           : styles.mainDefault;

     return (
       <div className={shellClass} data-shell-variant={variant}>
         <nav className={`site-nav ${styles.siteNav}`} aria-label="사이트">
           <a
             href="./"
             className={styles.brandLink}
             aria-current={activeNav === 'home' ? 'page' : undefined}
           >
             ⌂ HOME
           </a>
           <span className={`links ${styles.navLinks}`}>
             <a
               href="chords.html"
               className={styles.navItem}
               aria-current={activeNav === 'chords' ? 'page' : undefined}
             >
               CHORDS
             </a>
             <a
               href="log.html"
               className={styles.navItem}
               aria-current={activeNav === 'log' ? 'page' : undefined}
             >
               LOG
             </a>
             <a
               href="notes.html"
               className={styles.navItem}
               aria-current={activeNav === 'notes' ? 'page' : undefined}
             >
               NOTES
             </a>
             <a
               href="settings.html"
               className={styles.navItem}
               aria-current={activeNav === 'settings' ? 'page' : undefined}
             >
               SETTING
             </a>
             {showThemeToggle && (
               <span className={styles.themeGroup} role="group" aria-label="화면 모드">
                 <button
                   type="button"
                   id="theme-light"
                   className={styles.themeBtn}
                   aria-pressed={theme === 'light'}
                   onClick={() => handleThemePick('light')}
                 >
                   라이트
                 </button>
                 <button
                   type="button"
                   id="theme-dark"
                   className={styles.themeBtn}
                   aria-pressed={theme === 'dark'}
                   onClick={() => handleThemePick('dark')}
                 >
                   다크
                 </button>
               </span>
             )}
             <a
               href="settings.html#account"
               id="navAuth"
               data-nav-auth="1"
               data-auth-state={signedInUser ? 'signed-in' : 'signed-out'}
               className={styles.navItem}
               aria-label={navAriaLabel}
               target={!signedInUser && embedded ? '_blank' : undefined}
               rel={!signedInUser && embedded ? 'noopener' : undefined}
               onClick={handleAuthClick}
             >
               {navLabel}
             </a>
           </span>
         </nav>
         <main className={mainClass}>{children}</main>
       </div>
     );
   }
   ```

```ts
export const ALLOWED_SVG_ATTRS: ReadonlySet<string> = new Set([
  'xmlns', 'width', 'height', 'viewBox', 'font-family', 'transform',
  'fill', 'stroke', 'stroke-width', 'x', 'y', 'x1', 'y1', 'x2', 'y2',
  'cx', 'cy', 'r', 'd', 'font-size', 'font-weight', 'text-anchor',
]);

const SVG_NS = 'http://www.w3.org/2000/svg';

export function sanitizeSvgNode(srcNode: Node, doc: Document = document): Node | null {
  if (srcNode.nodeType === 3 /* Node.TEXT_NODE */) {
    return doc.createTextNode(srcNode.textContent || '');
  }
  if (srcNode.nodeType !== 1 /* Node.ELEMENT_NODE */) {
    return null;
  }
  const elem = srcNode as Element;
  const tag = (elem.localName || elem.tagName || '').toLowerCase();
  if (!ALLOWED_SVG_TAGS.has(tag)) {
    return null;
  }
  const cleanEl = doc.createElementNS(SVG_NS, tag);
  for (let i = 0; i < elem.attributes.length; i++) {
    const attr = elem.attributes[i];
    const name = attr.name.toLowerCase();
    if (ALLOWED_SVG_ATTRS.has(name)) {
      const val = attr.value;
      if (!val.toLowerCase().includes('javascript:') && !name.startsWith('on')) {
        cleanEl.setAttribute(attr.name, val);
      }
    }
  }
  for (let i = 0; i < elem.childNodes.length; i++) {
    const cleanChild = sanitizeSvgNode(elem.childNodes[i], doc);
    if (cleanChild) {
      cleanEl.appendChild(cleanChild);
    }
  }
  return cleanEl;
}

export function renderSanitizedSvg(container: Element | null, svgText: string | null | undefined): void {
  if (!container || typeof svgText !== 'string' || !svgText.trim()) return;
  try {
    const parser = new DOMParser();
    const parsedDoc = parser.parseFromString(svgText, 'image/svg+xml');
    if (parsedDoc.querySelector('parsererror')) return;
    const root = parsedDoc.documentElement;
    if (!root) return;
    const cleanEl = sanitizeSvgNode(root, container.ownerDocument || document);
    if (cleanEl) {
      container.appendChild(cleanEl);
    }
  } catch {
    /* 잘못된 SVG XML 무시 */
  }
}
```

```ts
  let lastFetchedUid: string | null = null;
  const listeners = new Set<(state: AuthState) => void>();

    return !!client;

  function isEmbedded(): boolean {
    try {
      return typeof window !== 'undefined' && window.self !== window.top;

      return true;

  function stripHash(url?: string): string {
    const fallback = typeof location !== 'undefined' ? location.href : '';
    return String(url || fallback).split('#')[0];

  function clearKeysCache(): void {
    try {
      localStorage.removeItem('ulp:keys');
    } catch {
      /* 저장 불가 무시 */
    }

  function saveKeysCache(me: MeProfile | null): void {
    if (!me || !me.id) return;
    try {
      const payload = {
        uid: me.id,
        notion: me.keys && me.keys.notion ? me.keys.notion : null,
        gemini: me.keys && me.keys.gemini ? me.keys.gemini : null,

    } catch {
      /* 저장 불가 무시 */
    }

  function computeInitial(user?: SupabaseUserLike | null, me?: MeProfile | null): string {
    const rawName = (
      (me && me.displayName) ||
      (user && user.user_metadata && (user.user_metadata.full_name || user.user_metadata.name)) ||
      (user && user.email ? user.email.split('@')[0] : '') ||
      ''
    ).trim();
    if (!rawName) return 'ME';
    if (/^[가-힣]/.test(rawName)) return rawName.slice(0, 2);
    const parts = rawName.split(/\s+/).filter(Boolean);
    if (parts.length >= 2 && /^[A-Za-z]/.test(parts[0]) && /^[A-Za-z]/.test(parts[1])) {
      return (parts[0][0] + parts[1][0]).toUpperCase();

    return rawName.slice(0, 2).toUpperCase();

      return currentSession ? currentSession.access_token : '';
    } catch {
      return currentSession ? currentSession.access_token : '';

    const options = opts || {};
    const token = await getAccessToken();
    const headers: Record<string, string> = Object.assign({}, (options.headers as Record<string, string>) || {});
    if (token) headers.Authorization = 'Bearer ' + token;
    if (options.body && typeof options.body === 'string' && !headers['Content-Type']) {
      headers['Content-Type'] = 'application/json; charset=utf-8';

      if (data && data.code === 'PROVIDER_KEY_INVALID') {
        clearKeysCache();
      }
      const msg = (data && (data.message || data.error)) || '요청 처리에 실패했어요 (' + res.status + ')';
      const err: ApiClientError = new Error(msg);
      err.status = res.status;
      err.code = data && data.code ? data.code : '';
      err.data = data;
      throw err;
    }
    return data as T;
  }

  async function fetchMe(): Promise<MeProfile | null> {
    if (!currentSession || !currentSession.access_token) {
      currentMe = null;
      lastFetchedUid = null;
      clearKeysCache();
      notify();
      return null;

      notify();

      fn(getSnapshot());
    } catch {

    }

      listeners.delete(fn);
    };

    if (!client) {
      notify();
      return;
    }

        event === 'USER_UPDATED' ||
        (event === 'SIGNED_IN' && (!currentMe || lastFetchedUid !== nextUid)) ||
        Boolean(nextUid && lastFetchedUid !== nextUid);

      if (shouldFetch) {
        void fetchMe();
      }
    });
  }

  };
```
```ts
}

function saveProgress(
  body: SaveProgressInput,
  opts?: { keepalive?: boolean }
): Promise<SaveProgressResponse> {
  try {
    requireLogin();
  } catch (e) {
    return Promise.reject(e);
  }
  const input = body || ({} as SaveProgressInput);
  const options = opts || {};
  const payload = {
    songId: input.songId,
    pos: input.pos,
    a: input.a,
    b: input.b,
    rate: input.rate,
    date: input.date,
    addSeconds: input.addSeconds,
    addLoops: input.addLoops,
  };
  const jsonBody = JSON.stringify(payload);
  const A = getAuthRef();

  if (options.keepalive === true) {
    const sess = A.getSession();
    const token = sess ? sess.access_token : '';
    const win: Record<string, any> = typeof window !== 'undefined' ? (window as any) : {};
    const apiBaseUrl = (win.ULEVELP_CONFIG && win.ULEVELP_CONFIG.apiBaseUrl) || '';
    const url = (apiBaseUrl ? String(apiBaseUrl).replace(/\/+$/, '') : '') + '/api/v1/practice/save';
    return fetch(url, {
      method: 'POST',
      keepalive: true,
      headers: {
        Authorization: 'Bearer ' + token,
        'Content-Type': 'application/json; charset=utf-8',
      },
      body: jsonBody,
    }).then(async (res) => {
      const data = res.status === 204 ? null : await res.json().catch(() => null);
      if (!res.ok) {
        const msg = (data && (data.message || data.error)) || '요청 처리에 실패했어요 (' + res.status + ')';
        const err: ApiClientError = new Error(msg);
        err.status = res.status;
        err.code = data && data.code ? data.code : '';
        err.data = data;
        throw err;
      }
      return data as SaveProgressResponse;
    });
  }

  return A.apiFetch('/api/v1/practice/save', {
    method: 'POST',
    body: jsonBody,
  });
}

function readLocal(prefix: string) {
  const out: Array<{
    videoId: string;
    pos: number | null;
    a: number | null;
    b: number | null;
    rate: number | null;
    date: string | null;
  }> = [];
  try {
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i);
      if (!k || !k.startsWith(prefix)) continue;
      const vid = k.slice(prefix.length);
      if (!/^[\w-]{11}$/.test(vid)) continue;
      let parsed: any = null;
      try {
        const raw = localStorage.getItem(k);
        parsed = raw ? JSON.parse(raw) : null;
      } catch {
        continue;
      }
      if (!parsed || typeof parsed !== 'object') continue;
      const pos = typeof parsed.pos === 'number' && isFinite(parsed.pos) ? parsed.pos : null;
      const a = typeof parsed.a === 'number' && isFinite(parsed.a) ? parsed.a : null;
      const b = typeof parsed.b === 'number' && isFinite(parsed.b) ? parsed.b : null;
      const rate = typeof parsed.rate === 'number' && isFinite(parsed.rate) ? parsed.rate : null;
      const date = typeof parsed.date === 'string' ? parsed.date : null;
      out.push({ videoId: vid, pos, a, b, rate, date });
    }
  } catch {
    return [];
  }
  return out;
}

async function runLocalMigration(): Promise<LocalMigrationResult> {
  if (!isLoggedIn()) {
    return { status: 'skipped', created: 0, skipped: 0, message: '' };
  }
  const A = getAuthRef();
  const me = A && typeof A.getMe === 'function' ? A.getMe() : null;
  if (!me || me.migratedLocalAt != null || migrating) {
    return { status: 'skipped', created: 0, skipped: 0, message: '' };
  }

  const own = readLocal('ulp:v:');
  const sister = readLocal('pp:v:');
  const ownVids = new Set(own.map((it) => it.videoId));
  const filteredSister = sister.filter((it) => !ownVids.has(it.videoId));

  if (own.length === 0 && filteredSister.length === 0) {
    return { status: 'skipped', created: 0, skipped: 0, message: '' };
  }

  migrating = true;
  try {
    const items: typeof own = [];
    const confirmFn = typeof window !== 'undefined' && typeof window.confirm === 'function' ? window.confirm : () => true;
    if (own.length > 0) {
      const ok = confirmFn('이 기기에서 연습하던 기록 ' + own.length + '곡을 계정으로 옮길까요?');
      if (ok) items.push(...own);
    }
    if (filteredSister.length > 0) {
      const ok = confirmFn(
        '같은 기기의 guitar-practice 기록 ' + filteredSister.length + '곡도 가져올까요? (원래 기록은 그대로 남아요)'
      );
      if (ok) items.push(...filteredSister);
    }

    items.sort((x, y) => {
      if (x.date && y.date) return y.date.localeCompare(x.date);
      if (x.date) return -1;
      if (y.date) return 1;
      return 0;
    });
    const topItems = items.slice(0, 100);

    const r = await A.apiFetch('/api/v1/songs/migrate-local', {
      method: 'POST',
      body: JSON.stringify({ items: topItems, markMigrated: true }),
    });

    if (A && typeof A.fetchMe === 'function') {
      await A.fetchMe();
    }
    return {
      status: 'done',
      created: r && typeof r.created === 'number' ? r.created : 0,
      skipped: r && typeof r.skipped === 'number' ? r.skipped : 0,
      message: '',
    };
  } catch (e: any) {
    return {
      status: 'failed',
      created: 0,
      skipped: 0,
      message: (e && e.message) || '기기 기록을 옮기지 못했어요',
    };
  } finally {
    migrating = false;
  }
}

async function exportZip(includeAudio?: boolean): Promise<{ blob: Blob; filename: string }> {
  requireLogin();
  const A = getAuthRef();
  const sess = A && typeof A.getSession === 'function' ? A.getSession() : null;
  const token = sess && sess.access_token ? sess.access_token : '';
  const qs = '?includeAudio=' + (includeAudio !== false);
  const res = await fetch('/api/v1/export/zip' + qs, {
    headers: {
      Authorization: 'Bearer ' + token,
    },
  });
  if (!res.ok) {
    let errData: any = {};
    try {
      errData = await res.json();
    } catch {
      /* ignore */
    }
    const msg = (errData && (errData.message || errData.error)) || '내보내기에 실패했어요 (' + res.status + ')';
    const err: ApiClientError = new Error(msg);
    err.status = res.status;
    err.code = (errData && errData.code) || 'EXPORT_ERROR';
    err.data = errData;
    throw err;
  }
  const cd = res.headers.get('content-disposition') || '';
  let filename = 'ULeveLP-export.zip';
  const match = cd.match(/filename="?([^"]+)"?/);
  if (match && match[1]) {
    filename = match[1];
  }
  const blob = await res.blob();
  return { blob, filename };
}
```

```ts
  async function uploadBlob(blob: Blob, name?: string, mime?: string, folderId?: string | null): Promise<string> {
    if (!blob) throw new Error('업로드할 파일이 없습니다.');
    const token = await requestToken();
    const contentType = mime || 'audio/mp4';
    const fileName = name || 'take-' + Date.now() + '.m4a';
    const size = blob.size;

    if (size <= 5 * 1024 * 1024) {
      const boundary = '-------ulp_boundary_' + Math.random().toString(36).slice(2);
      const meta = JSON.stringify({
        name: fileName,
        parents: folderId ? [folderId] : [],
        mimeType: contentType,

      const body = new Blob(

          '--' + boundary + '
Content-Type: application/json; charset=UTF-8

' + meta + '
--' + boundary + '
Content-Type: ' + contentType + '

',

          '
--' + boundary + '--',
        ],
        { type: 'multipart/related; boundary=' + boundary }
      );

      const res = await fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id', {
        method: 'POST',
        headers: {
          Authorization: 'Bearer ' + token,
        },
        body,
      });

      if (!res.ok) {
        let errData: any = {};

          errData = await res.json();
        } catch {
          /* ignore */
        }

      }
      const data = await res.json();
      return data.id;
    }

      method: 'POST',
      headers: {
        Authorization: 'Bearer ' + token,
        'Content-Type': 'application/json; charset=UTF-8',

      throw new Error('Drive 세션 시작 실패 (' + initRes.status + ')');
```

```ts
export function createAudioEngine(ctxProvider?: () => any): AudioEngine {
  let ctx: any = null;

  function ensureContext(): any {
    if (ctx) return ctx;
    if (typeof ctxProvider === 'function') {
      ctx = ctxProvider();
      return ctx;
    }
    if (typeof window !== 'undefined') {
      const AC = (window as any).AudioContext || (window as any).webkitAudioContext;
      if (AC) ctx = new AC();
    }
    return ctx;
  }

  async function resume(): Promise<void> {
    const c = ensureContext();
    if (c && c.state === 'suspended' && typeof c.resume === 'function') {
      await c.resume();
    }
  }

  function pluck(midi: number, when?: number, dur = 0.85): void {
    const c = ensureContext();
    if (!c) return;
    const t0 = typeof when === 'number' ? when : c.currentTime;
    const f = midiToFreq(midi);
    const o1 = c.createOscillator();
    const o2 = c.createOscillator();
    const g = c.createGain();
    o1.type = 'triangle';
    o1.frequency.setValueAtTime(f, t0);
    o2.type = 'sine';
    o2.frequency.setValueAtTime(f * 2, t0);
    const g2 = c.createGain();
    g2.gain.setValueAtTime(0.18, t0);
    g.gain.setValueAtTime(0.0001, t0);
    g.gain.exponentialRampToValueAtTime(0.22, t0 + 0.01);
    g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
    o1.connect(g);
    o2.connect(g2);
    g2.connect(g);
    g.connect(c.destination);
    o1.start(t0);
    o2.start(t0);
    o1.stop(t0 + dur + 0.02);
    o2.stop(t0 + dur + 0.02);
  }

  function bassPluck(midi: number, when?: number, dur = 1.05): void {
    const c = ensureContext();
    if (!c) return;
    const t0 = typeof when === 'number' ? when : c.currentTime;
    const f = midiToFreq(midi);
    const oFund = c.createOscillator();
    const oHarm = c.createOscillator();
    const gHarm = c.createGain();
    const gMain = c.createGain();

    oFund.type = 'triangle';
    oFund.frequency.setValueAtTime(f, t0);
    oHarm.type = 'sine';
    oHarm.frequency.setValueAtTime(f * 2, t0);
    gHarm.gain.setValueAtTime(0.28, t0);

    gMain.gain.setValueAtTime(0.0001, t0);
    gMain.gain.exponentialRampToValueAtTime(0.32, t0 + 0.012);
    gMain.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);

    oFund.connect(gMain);
    oHarm.connect(gHarm);
    gHarm.connect(gMain);
    gMain.connect(c.destination);

    oFund.start(t0);
    oHarm.start(t0);
    oFund.stop(t0 + dur + 0.02);
    oHarm.stop(t0 + dur + 0.02);
  }

  function chord(midis: number[], when?: number, dur = 1.4): void {
    const c = ensureContext();
    if (!c) return;
    const t0 = typeof when === 'number' ? when : c.currentTime;
    midis
      .slice()
      .sort((a, b) => a - b)
      .forEach((m, i) => pluck(m, t0 + i * 0.035, dur));
  }

  function doubleStop(midis: [number, number], when?: number, dur = 1.1, isBass = false): void {
    const c = ensureContext();
    if (!c) return;
    const t0 = typeof when === 'number' ? when : c.currentTime;
    const fn = isBass ? bassPluck : pluck;
    fn(midis[0], t0, dur);
    fn(midis[1], t0, dur);
  }

  return {
    resume,
    pluck,
    bassPluck,
    chord,
    doubleStop,
  };
}

export const audioEngine = createAudioEngine();
```

   ```tsx
   import React, { useEffect, useRef } from 'react';
   import { safeAudioUrl } from '../../lib/security';

   export interface SafeAudioProps extends Omit<React.AudioHTMLAttributes<HTMLAudioElement>, 'src' | 'style'> {
     srcUrl: string | null | undefined;
     takeId?: string;
   }

   export function SafeAudio({
     srcUrl,
     takeId,
     controls = true,
     preload = 'metadata',
     ...rest
   }: SafeAudioProps): React.ReactElement {
     const audioRef = useRef<HTMLAudioElement | null>(null);

     useEffect(() => {
       const el = audioRef.current;
       if (!el) return;
       const valid = safeAudioUrl(srcUrl);
       if (!valid) {
         el.removeAttribute('src');
         delete el.dataset.loadedSrc;
         delete el.dataset.takeId;
         try {
           el.load();
         } catch {
           /* JSDOM/비브라우저 환경 무시 */
         }
         return;
       }
       const prevSrc = el.dataset.loadedSrc || '';
       const prevId = el.dataset.takeId || '';
       const nextId = takeId || valid;
       if (prevSrc !== valid || prevId !== nextId) {
         el['src'] = valid;
         el.dataset.loadedSrc = valid;
         el.dataset.takeId = nextId;
       }
     }, [srcUrl, takeId]);

     return <audio ref={audioRef} controls={controls} preload={preload} {...rest} />;
   }
   ```

```js
function extractCssVars(cssBlock) {
  const vars = new Map();
  const re = /(--[a-zA-Z0-9-]+)\s*:\s*([^;]+);/g;
  for (const m of cssBlock.matchAll(re)) {
    vars.set(m[1].trim(), m[2].trim());
  }
  return vars;
}

const REQUIRED_TOKENS = [
  '--bg', '--surface', '--surface2', '--ink', '--ink-navy', '--muted', '--line',
  '--accent', '--accent-ink', '--blue-vivid', '--blue-deep', '--accent-text', '--link',

test('R0-T1: tokens.css ↔ public/styles.css 필수 디자인 토큰 일치 및 순백(#ffffff/#fff) 0건', () => {
  const tokensCss = readText('frontend/src/styles/tokens.css');
  const stylesCss = readText('public/styles.css');
  const siteCss = readText('public/site.css');
  const practiceCss = readText('public/practice.css');

  const rootTokens = extractCssVars(tokensCss.split('@media')[0]);
  const rootStyles = extractCssVars(stylesCss.split('@media')[0]);

  for (const token of REQUIRED_TOKENS) {
    assert.ok(rootTokens.has(token), 'tokens.css 누락 토큰: ' + token);
    assert.ok(rootStyles.has(token), 'public/styles.css 누락 토큰: ' + token);
    assert.equal(
      rootTokens.get(token).toLowerCase(),
      rootStyles.get(token).toLowerCase(),
      '토큰 값 불일치: ' + token
    );
  }

  assert.equal(rootTokens.get('--bg').toUpperCase(), '#F0EEEA');
  assert.equal(rootTokens.get('--radius-pill'), '9999px');
  assert.equal(rootTokens.get('--radius-card'), '0px');
  assert.equal(rootTokens.get('--container-max'), '1120px');

  // 주석 제거 후 #ffffff 또는 #fff 존재 여부 검사
  for (const [name, content] of [
    ['frontend/src/styles/tokens.css', tokensCss],
    ['public/styles.css', stylesCss],
    ['public/site.css', siteCss],
    ['public/practice.css', practiceCss],
  ]) {
    const noComments = content.replace(/\/\*[\s\S]*?\*\//g, '');
    const whiteMatch = noComments.match(/#(?:ffffff|fff)/i);
    assert.equal(whiteMatch, null, name + '에 금지된 순백 컬러가 남아 있음: ' + (whiteMatch && whiteMatch[0]));
  }
});

test('R0-T4: frontend/src 내 인라인 style={{ 및 dangerouslySetInnerHTML 0건, frontend/index.html CSP 일치', () => {
  function walkTs(dir, out = []) {
    for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
      const p = path.join(dir, e.name);
      if (e.isDirectory()) walkTs(p, out);
      else if (/\.(?:ts|tsx|html|css)$/.test(e.name)) out.push(p);
    }
    return out;
  }

  const files = walkTs(path.join(ROOT, 'frontend'));
  assert.ok(files.length >= 10, 'frontend/ 파일이 충분히 생성되어야 함');

  for (const f of files) {
    const rel = path.relative(ROOT, f).split(path.sep).join('/');
    const txt = fs.readFileSync(f, 'utf8');
    assert.equal(/style\s*=\s*\{\{/.test(txt), false, rel + '에 금지된 React 인라인 style={{ 속성이 있음');
    assert.equal(/dangerouslySetInnerHTML/.test(txt), false, rel + '에 금지된 dangerouslySetInnerHTML이 있음');
  }

  const indexHtml = readText('frontend/index.html');
  const pubIndexHtml = readText('public/index.html');
  const meta1 = indexHtml.match(/<meta\s+http-equiv="Content-Security-Policy"\s+content="([^"]*)"/i);

});
```

## 9. 금지 사항

1. **파일 목록 밖 수정 금지:** 1절 파일 목록에 없는 파일(`src/main/kotlin/**`, `server.js`, `public/app.js`, `public/home.js`, `public/log.js`, `public/notes.js`, `public/note.js`, `public/practice.js`, `public/settings.js` 등)은 절대 수정하지 않는다.
2. **Vite 빌드 산출물을 `public/`에 출력 금지:** `vite.config.ts`의 `build.outDir`은 반드시 `dist`로 설정하며, `public/` 폴더에 번들 파일을 덮어써서 `test/security.test.js`를 깨뜨리지 않는다.
3. **인라인 `style` 속성 및 `<style>` 태그 절대 금지:** CSP에 `'unsafe-inline'`이 없으므로 React 컴포넌트에서 `style={{ ... }}` prop이나 `el.setAttribute('style', ...)`을 절대 쓰지 않는다.
4. **`innerHTML` / `dangerouslySetInnerHTML` 사용 금지:** SVG 마운트는 반드시 `<SanitizedSvg />` (`renderSanitizedSvg`)만 사용한다.
5. **순백(`#FFFFFF`, `#fff`) 및 중간 곡률(`3px`, `4px`, `6px`, `8px`, `12px`, `16px`, `999px`) 사용 금지:** 밝은 표면/글자는 `#F6F6F1`(`var(--surface)` / `var(--root-ink)`)만 쓰고, 곡률은 `var(--radius-pill)`(`9999px`)과 `var(--radius-card)`(`0px`) 두 가지만 사용한다.
6. **`public/data.js` 및 `apiClient.ts`에서 `replaceOldSongTakes` 무조건 포함 금지:** `meta.replaceOldSongTakes === true`일 때만 키를 추가하여 기본 호출 시 기존 8개 키 단언 테스트(`D5`)가 깨지지 않게 한다.
7. **`PUT /api/v1/integrations/gemini` 응답을 `{ integration: ... }`으로 감싸 읽지 않는다:** 서버(`IntegrationRoutes.kt:83`)는 루트 레벨 `IntegrationDto`(`{ provider, hint, config, updatedAt, verified, displayName }`)를 반환하므로 루트 필드를 그대로 읽는다.
8. **오디오 `src`에 빈 문자열(`""`) 대입 금지:** `<SafeAudio />`에서 빈 URL일 때는 반드시 `removeAttribute('src')`와 `load()`를 호출한다(`주니어_체크리스트.md` 57행).
9. **`test/r0_core.test.js`에서 외부 `node_modules` 패키지 top-level `require` 금지:** Node 내장 모듈(`node:test`, `node:assert/strict`, `node:fs`, `node:path`, `node:vm`)만 사용한다.
10. **연속 작업 금지 (`협업_시니어주니어.md` 3·8절 및 `GEMINI.md` 관문 3):** `R0-1`은 묶음 `R0`의 마지막 step이므로, `R0-1` 완료 후 절대 `R1-1`을 이어서 진행하지 않고 즉시 관문 문구(`"R0 묶음(R0-1) 단계 완료하였습니다. 다음 진행은 문서 보강 및 검토 후 진행이 필요합니다."`)를 출력하고 멈춘다.
