export function getLocalTodayStr(now: Date = new Date()): string {
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, '0');
  const d = String(now.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

function parseYmdUtc(ymd: string): Date | null {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(ymd || '').trim());
  if (!m) return null;
  const year = Number(m[1]);
  const month = Number(m[2]) - 1;
  const day = Number(m[3]);
  const dt = new Date(Date.UTC(year, month, day));
  if (
    dt.getUTCFullYear() !== year ||
    dt.getUTCMonth() !== month ||
    dt.getUTCDate() !== day
  ) {
    return null;
  }
  return dt;
}

function formatYmdUtc(dt: Date): string {
  const y = dt.getUTCFullYear();
  const m = String(dt.getUTCMonth() + 1).padStart(2, '0');
  const d = String(dt.getUTCDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

function addDaysUtc(dt: Date, deltaDays: number): Date {
  return new Date(dt.getTime() + deltaDays * 86400000);
}

/**
 * 이번 주 월(Mo)~일(Su) 7일 스트릭 현황과 작심삼일 연속 일수를 계산한다.
 * @param activeDates 연습 기록이 존재하는 YYYY-MM-DD 문자열 목록
 * @param todayStr 기준 오늘 날짜 (YYYY-MM-DD, 기본값 로컬 오늘)
 */
export function computeWeeklyStreak(
  activeDates: Iterable<string>,
  todayStr: string = getLocalTodayStr()
): WeeklyStreakSummary {
  const practicedSet = new Set<string>();
  for (const d of activeDates) {
    if (typeof d === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(d.trim())) {
      practicedSet.add(d.trim());
    }
  }

  const baseDate = parseYmdUtc(todayStr) || parseYmdUtc(getLocalTodayStr())!;
  const cleanToday = formatYmdUtc(baseDate);

  // 월요일(Mo=0) ~ 일요일(Su=6) 오프셋
  const jsDay = baseDate.getUTCDay(); // 0(일) ~ 6(토)
  const mondayOffset = jsDay === 0 ? -6 : 1 - jsDay;
  const mondayDate = addDaysUtc(baseDate, mondayOffset);

  const days: WeeklyStreakDay[] = DOW_LABELS.map((dowLabel, idx) => {
    const dt = addDaysUtc(mondayDate, idx);
    const dateStr = formatYmdUtc(dt);
    return {
      dowLabel,
      dateStr,
      dayNumber: dt.getUTCDate(),
      practiced: practicedSet.has(dateStr),
      isToday: dateStr === cleanToday,
    };
  });

  // 연속 연습일(streakDays): 오늘 연습했으면 오늘부터 역산, 아직 안 했으면 어제부터 역산
  let streakDays = 0;
  const yesterdayStr = formatYmdUtc(addDaysUtc(baseDate, -1));
  let cursor: Date | null = null;
  if (practicedSet.has(cleanToday)) {
    cursor = baseDate;
  } else if (practicedSet.has(yesterdayStr)) {
    cursor = addDaysUtc(baseDate, -1);
  }

  while (cursor && streakDays < 3650) {
    const curStr = formatYmdUtc(cursor);
    if (!practicedSet.has(curStr)) break;
    streakDays++;
    cursor = addDaysUtc(cursor, -1);
  }

  const cycleStep = streakDays === 0 ? 0 : ((streakDays - 1) % 3) + 1;
  const challengeText =
    streakDays === 0
      ? '작심삼일 도전하기 · 오늘 첫 연습을 시작해 보세요'
      : `작심삼일 도전하기 · 연속 ${streakDays}일째 (${cycleStep}/3)`;

  return {
    days,
    streakDays,
    cycleStep,
    challengeText,
  };
}

/** 초(number)를 MM:SS 문자열로 포맷한다 (예: 90 -> "01:30") */
export function formatSec(sec: number | null | undefined): string {
  if (typeof sec !== 'number' || !Number.isFinite(sec) || sec < 0) return '00:00';
  const total = Math.floor(sec);
  const m = Math.floor(total / 60);
  const s = total % 60;
  return String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0');
}

/** 최근 연습 곡의 상태/구간 요약 문구를 만든다 */
export function formatRecentSongSegment(song: {
  a?: number | null;
  b?: number | null;
  pos?: number | null;
  rate?: number | null;
}): string {
  if (
    typeof song.a === 'number' &&
    typeof song.b === 'number' &&
    Number.isFinite(song.a) &&
    Number.isFinite(song.b) &&
    song.b > song.a
  ) {
    const ratePart = song.rate && song.rate !== 1 ? ` · ${song.rate}×` : '';
    return `${formatSec(song.a)} ~ ${formatSec(song.b)} 구간 연습 중${ratePart}`;
  }
  const posStr = typeof song.pos === 'number' && song.pos > 0 ? formatSec(song.pos) : '00:00';
  const rateStr = song.rate ? `${song.rate}×` : '1×';
  return `마지막 위치 ${posStr} · ${rateStr} 연습 중`;
}

/** 최근 연습 곡의 플레이어 이어하기 상대 경로를 생성한다 */
export function buildPracticeResumeHref(song: {
  videoId: string;
  pos?: number | null;
  a?: number | null;
  b?: number | null;
  rate?: number | null;
}): string {
  const vid = videoIdFrom(song.videoId) || '';
  if (!vid) return 'practice.html';
  const params = new URLSearchParams();
  params.set('v', vid);
  if (typeof song.pos === 'number' && Number.isFinite(song.pos) && song.pos > 0) {
    params.set('t', String(Math.floor(song.pos)));
  }
  if (
    typeof song.a === 'number' &&
    typeof song.b === 'number' &&
    Number.isFinite(song.a) &&
    Number.isFinite(song.b) &&
    song.b > song.a
  ) {
    params.set('a', String(song.a));
    params.set('b', String(song.b));
  }
  if (typeof song.rate === 'number' && Number.isFinite(song.rate) && song.rate !== 1) {
    params.set('r', String(song.rate));
  }
  return 'practice.html?' + params.toString();
}

/** 현재 인덱스와 다른 다음 인덱스를 결정론적/난수로 반환한다 */
export function pickNextIndex(currentIndex: number, length: number, randFn: () => number = Math.random): number {
  if (length <= 1) return 0;
  const rawPick = Math.floor(randFn() * (length - 1));
  const clamped = Math.max(0, Math.min(length - 2, rawPick));
  return clamped >= currentIndex ? clamped + 1 : clamped;
}

/** 노트 상세/요약 목록 및 로컬 캐시에서 REFERENCE / ARCHIVE 진행 목록을 구성한다 */
export function extractProgressionsFromNotes(
  notes: ReadonlyArray<NoteSummaryDto | NoteDetailDto>
): { reference: ProgressionItem[]; archive: ProgressionItem[] } {
  const customRef: ProgressionItem[] = [];
  const customArch: ProgressionItem[] = [];

  const stored = json.get<ProgressionItem[]>('ulp:progressions', []);
  if (Array.isArray(stored)) {
    for (const item of stored) {
      if (!item || typeof item.chordsText !== 'string' || !item.chordsText.trim()) continue;
      if (item.category === 'REFERENCE') customRef.push(item);
      else if (item.category === 'ARCHIVE') customArch.push(item);
    }
  }

  for (const n of notes || []) {
    if (!n || !n.id) continue;
    const maybeDetail = n as NoteDetailDto;
    const progs =
      maybeDetail.analysis &&
      maybeDetail.analysis.harmony &&
      Array.isArray(maybeDetail.analysis.harmony.progressions)
        ? maybeDetail.analysis.harmony.progressions
        : [];

    if (progs.length > 0) {
      progs.forEach((p: any, idx: number) => {
        const chords = Array.isArray(p?.chords)
          ? p.chords.map((c: any) => String(c?.symbol || '').trim()).filter(Boolean)
          : [];
        if (chords.length >= 2) {
          const item: ProgressionItem = {
            id: `note-${n.id}-${idx}`,
            category: Array.isArray(n.tags) && n.tags.includes('편곡') ? 'REFERENCE' : 'ARCHIVE',
            chordsText: chords.join(' - '),
            title: String(p.name || n.title || '화성 노트 진행'),
            keyLabel: String(p.key || '분석 노트'),
            sourceType: 'AI',
            href: 'note.html?id=' + encodeURIComponent(n.id),
          };
          if (item.category === 'REFERENCE') customRef.push(item);
          else customArch.push(item);
        }
      });
    }
  }

  return {
    reference: customRef.length > 0 ? [...customRef, ...DEFAULT_REFERENCE_PROGRESSIONS] : [...DEFAULT_REFERENCE_PROGRESSIONS],
    archive: customArch.length > 0 ? [...customArch, ...DEFAULT_ARCHIVE_PROGRESSIONS] : [...DEFAULT_ARCHIVE_PROGRESSIONS],
  };
}

export function getLastChordTab(): string {
  const saved = raw.get('ulp:tab');
  return CHORD_MODES.some(([id]) => id === saved) ? saved : '';
}

const ROOT = path.join(__dirname, '..');

function readText(relPath) {
  return fs.readFileSync(path.join(ROOT, relPath), 'utf8').replace(/\r\n/g, '\n');
}

// TypeScript 소스에서 순수 런타임 검증을 위해 타입 어노테이션만 제거하는 경량 변환기
function transpileTsForVm(tsCode) {
  return tsCode
    .replace(/import\s+type\s+[^;]+;/g, '')
    .replace(/import\s+\{\s*([^}]+)\s*\}\s+from\s+['"]([^'"]+)['"];/g, 'const {$1} = __require("$2");')
    .replace(/export\s+type\s+[A-Za-z0-9_]+\s*=[\s\S]*?;/g, '')
    .replace(/export\s+interface\s+[A-Za-z0-9_]+\s*\{[\s\S]*?\n\}/g, '')
    .replace(/export\s+const\s+([A-Za-z0-9_]+)\s*:[^=]+=/g, 'exports.$1 =')
    .replace(/export\s+const\s+([A-Za-z0-9_]+)\s*=/g, 'exports.$1 =')
    .replace(/export\s+function\s+([A-Za-z0-9_]+)/g, 'exports.$1 = function $1')
    .replace(/:\s*ReadonlyArray<[^>]+>/g, '')
    .replace(/:\s*Iterable<[^>]+>/g, '')
    .replace(/:\s*ProgressionItem\[\]/g, '')
    .replace(/:\s*WeeklyStreakDay\[\]/g, '')
    .replace(/:\s*WeeklyStreakSummary/g, '')
    .replace(/:\s*Date\s*\|\s*null/g, '')
    .replace(/:\s*Date/g, '')
    .replace(/:\s*number\s*\|\s*null\s*\|\s*undefined/g, '')
    .replace(/:\s*string/g, '')
    .replace(/:\s*number/g, '')
    .replace(/:\s*boolean/g, '')
    .replace(/:\s*any/g, '')
    .replace(/:\s*\(\)\s*=>\s*number/g, '')
    .replace(/:\s*\{[^}]+\}/g, '')
    .replace(/as\s+NoteDetailDto/g, '')
    .replace(/json\.get<[^>]+>/g, 'json.get')
    .replace(/new\s+Set<string>\(\)/g, 'new Set()')
    .replace(/!\s*;/g, ';');
}

function loadProgressionsModule(storageMap = new Map()) {
  const secFake = {
    raw: {
      get: (k) => storageMap.get(k) || '',
      set: (k, v) => (v ? storageMap.set(k, v) : storageMap.delete(k)),
    },
    json: {
      get: (k, d) => (storageMap.has(k) ? JSON.parse(storageMap.get(k)) : d),
      set: (k, v) => storageMap.set(k, JSON.stringify(v)),
    },
    videoIdFrom: (s) => {
      const str = String(s ?? '').trim();
      if (/^[\w-]{11}$/.test(str)) return str;
      const m = str.match(/(?:youtu\.be\/|[?&]v=|\/(?:shorts|embed|live)\/)([\w-]{11})/);
      return m ? m[1] : null;
    },
  };
  const code = transpileTsForVm(readText('frontend/src/lib/progressions.ts'));
  const exportsObj = {};
  const ctx = vm.createContext({
    exports: exportsObj,
    __require: () => secFake,
    Date,
    Math,
    Number,
    String,
    Array,
    Set,
    URLSearchParams,
    encodeURIComponent,
  });
  vm.runInContext(code, ctx);
  return exportsObj;
}

test('R1-H1: computeWeeklyStreak — Mo~Su 7일 배열 생성 및 연속 연습일·작심삼일(0일/2일/3일/4일) 계산', () => {
  const P = loadProgressionsModule();

  // 1) 연습 기록 0건 (기준일: 2026-11-04 수요일)
  const s0 = P.computeWeeklyStreak([], '2026-11-04');
  assert.equal(s0.days.length, 7);
  assert.deepEqual(
    s0.days.map((d) => d.dowLabel),
    ['Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa', 'Su']
  );
  assert.equal(s0.days[0].dateStr, '2026-11-02');
  assert.equal(s0.days[2].dateStr, '2026-11-04');
  assert.equal(s0.days[2].isToday, true);
  assert.equal(s0.streakDays, 0);
  assert.equal(s0.cycleStep, 0);
  assert.equal(s0.challengeText, '작심삼일 도전하기 · 오늘 첫 연습을 시작해 보세요');

  // 2) 어제·오늘 2일 연속 연습
  const s2 = P.computeWeeklyStreak(['2026-11-03', '2026-11-04'], '2026-11-04');
  assert.equal(s2.streakDays, 2);
  assert.equal(s2.cycleStep, 2);
  assert.equal(s2.days[1].practiced, true);
  assert.equal(s2.days[2].practiced, true);
  assert.equal(s2.challengeText, '작심삼일 도전하기 · 연속 2일째 (2/3)');

  // 3) 오늘은 아직 연습 전이지만 어제까지 3일 연속 연습한 경우 스트릭 3일 유지
  const s3 = P.computeWeeklyStreak(['2026-11-01', '2026-11-02', '2026-11-03'], '2026-11-04');
  assert.equal(s3.streakDays, 3);
  assert.equal(s3.cycleStep, 3);
  assert.equal(s3.challengeText, '작심삼일 도전하기 · 연속 3일째 (3/3)');

  // 4) 4일 연속 연습 시 새 3일 사이클 1/3 시작
  const s4 = P.computeWeeklyStreak(['2026-11-01', '2026-11-02', '2026-11-03', '2026-11-04'], '2026-11-04');
  assert.equal(s4.streakDays, 4);
  assert.equal(s4.cycleStep, 1);
  assert.equal(s4.challengeText, '작심삼일 도전하기 · 연속 4일째 (1/3)');
});

test('R1-H3: pickNextIndex 비중복 보장 및 extractProgressionsFromNotes 노트 진행 병합', () => {
  const P = loadProgressionsModule();
  for (let cur = 0; cur < 5; cur++) {
    for (const r of [0, 0.25, 0.5, 0.75, 0.99]) {

    }

    {
      id: 'n-100',
      title: '네오소울 화성 강의',
      tags: ['화성'],
      analysis: {
        harmony: {

              key: 'Eb Major',
              chords: [{ symbol: 'Fm9' }, { symbol: 'Bb13' }, { symbol: 'EbM9' }],
            },
          ],
        },
      },
    },
  ]);
  assert.equal(extracted.archive[0].chordsText, 'Fm9 - Bb13 - EbM9');
  assert.equal(extracted.archive[0].sourceType, 'AI');
  assert.equal(extracted.archive[0].href, 'note.html?id=n-100');

});

### 8.2 금지 사항
1. **파일 목록 밖 수정 금지:** 1절 목록에 없는 파일(`public/settings.js`, `public/notes.js`, `public/note.js`, `public/app.js`, `test/security.test.js` 등)은 절대 수정하지 않는다.
2. **`public/home.js`에서 `localStorage.getItem(k)` 변수 키 직접 호출 금지:** 비로그인 기기 연습 곡(`ulp:v:*`)은 반드시 `ULData.listLocalSongs()`를 통해 조회하여 `test/security.test.js` `(g)` 규칙을 준수한다.
3. **`public/home.js`에서 `.href` / `.src`에 일반 식별자 변수 단독 대입 금지:** `test/security.test.js` `(d)` 규칙에 따라必ず 문자열 리터럴 + `encodeURIComponent(...)` 조합, `safeUrl(...)`, 또는 `thumb(...)` 직접 호출만 대입한다.
4. **인라인 `style` 속성, `dangerouslySetInnerHTML`, 순백(`#FFFFFF`/`#fff`), 중간 곡률(`3~16px`) 절대 금지:** 모든 스타일은 `HomePage.module.css`와 `public/site.css`에서 토큰 변수(`--radius-card: 0px`, `--radius-pill: 9999px`)로만 정의한다.
5. **기존 DOM ID 삭제·변경 금지:** `E1` 및 `csp_check.js`가 참조하는 `#go`, `#goUrl`, `#goPick`, `#goThumb`, `#goPlay`, `#goNote`, `#goMsg`, `#modes`, `#songs`, `#notionBig`, `#notionLinks`, `#nlPractice`, `#nlNotes`, `#copy` ID를 그대로 유지한다.
6. **연속 작업 금지:** 한 세션에서는 `R1-1`만 완결하고, `R1-2`는 다음 세션 시작 지시로 안내한다.
